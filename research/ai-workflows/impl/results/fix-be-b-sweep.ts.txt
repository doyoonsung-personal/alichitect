// Offline audit of canonical fixtures and stored review designs; output goes to stdout.
import { readFileSync, readdirSync } from 'node:fs';
import { loadCatalog, loadModels } from '../../packages/core/src/catalog-node.ts';
import { checkWorkflow, validateWorkflowOp } from '../../packages/core/src/workflow-validate.ts';
import type { WorkflowGraph, WorkflowOp, Warning } from '../../packages/core/src/types.ts';
import { applyWorkflowOp, createWorkflowCtx, emptyWorkflow } from '../../packages/core/src/workflow.ts';
const catalog = loadCatalog(), models = loadModels();
const root = 'research/ai-workflows/impl/review-input/eval-results';
const result: Record<string, { validation: Warning[]; checks: Warning[] }> = {};
function audit(key: string, ops: unknown[], stored?: WorkflowGraph) {
  let workflow = emptyWorkflow();
  const ctx = createWorkflowCtx();
  const validation: Warning[] = [];
  const validationCtx = {};
  for (const raw of ops) {
    const r = validateWorkflowOp(raw, catalog, models, validationCtx);
    validation.push(...r.warnings);
    if (r.op && ['step', 'link', 'splice', 'each', 'unstep'].includes(r.op.t)) {
      workflow = applyWorkflowOp(workflow, r.op as WorkflowOp, ctx).workflow;
    }
  }
  result[key] = { validation, checks: checkWorkflow(stored ?? workflow, catalog, models) };
}
for (const file of readdirSync('packages/core/test/fixtures/workflows').filter(f => f.endsWith('.json'))) {
  const fixture = JSON.parse(readFileSync(`packages/core/test/fixtures/workflows/${file}`, 'utf8'));
  audit(file, fixture.ops);
}
for (const dir of readdirSync(root)) {
  for (const line of readFileSync(`${root}/${dir}/runs.jsonl`, 'utf8').trim().split('\n')) {
    const run = JSON.parse(line);
    for (const turn of run.turns) if (turn.workflow) audit(`${dir}/${run.id}/${turn.index}`, turn.ops, turn.workflow);
  }
}
console.log(JSON.stringify(result, null, 2));
